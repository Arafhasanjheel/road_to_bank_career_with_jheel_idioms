        let currentCardIndex = 0;

        function renderList(items) {
            const listContainer = document.getElementById('idiom-list');
            listContainer.innerHTML = '';
            items.forEach(item => {
                const card = document.createElement('div');
                card.className = 'idiom-card';
                card.innerHTML = `
                    <div class="idiom-header">
                        <span class="idiom-title">${item.text}</span>
                        <span class="idiom-num">#${item.sl}</span>
                    </div>
                    <div class="idiom-meaning">${item.meaning}</div>
                    <div class="idiom-sentence">Ex: "${item.sentence}"</div>
                `;
                listContainer.appendChild(card);
            });
        }

        function filterIdioms() {
            const query = document.getElementById('search').value.toLowerCase();
            const filtered = idioms.filter(i => 
                i.text.toLowerCase().includes(query) || 
                i.meaning.toLowerCase().includes(query)
            );
            renderList(filtered);
        }

        function switchTab(tab) {
            document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));
            document.querySelectorAll('.tab-content').forEach(c => c.classList.remove('active'));
            
            if(tab === 'list') {
                document.querySelectorAll('.tab-btn')[0].classList.add('active');
                document.getElementById('list-tab').classList.add('active');
            } else {
                document.querySelectorAll('.tab-btn')[1].classList.add('active');
                document.getElementById('flashcard-tab').classList.add('active');
                updateCard();
            }
        }

        function updateCard() {
            const card = idioms[currentCardIndex];
            document.getElementById('flashcard').classList.remove('flipped');
            document.getElementById('card-idiom').innerText = card.text;
            document.getElementById('card-meaning').innerText = card.meaning;
            document.getElementById('card-sentence').innerText = `Ex: "${card.sentence}"`;
            document.getElementById('card-progress').innerText = `Card ${currentCardIndex + 1} of ${idioms.length}`;
        }

        function flipCard() {
            document.getElementById('flashcard').classList.toggle('flipped');
        }

        function nextCard() {
            if(currentCardIndex < idioms.length - 1) {
                currentCardIndex++;
                updateCard();
            }
        }

        function prevCard() {
            if(currentCardIndex > 0) {
                currentCardIndex--;
                updateCard();
            }
        }

        // Initialize List
        renderList(idioms);
    </script>
