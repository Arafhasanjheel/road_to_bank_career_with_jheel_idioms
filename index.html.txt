    <script>
        const wordsData = [
            { id: 1, category: 'editorial', normal: 'Necessary (প্রয়োজনীয়)', pro: 'Necessitate / Require', meaning: 'প্রয়োজন হওয়া বা দরকার হওয়া', example: 'Controlling this problem necessitates immediate action.' },
            { id: 2, category: 'business', normal: 'Help (সাহায্য করা)', pro: 'Facilitate / Assist', meaning: 'সহজতর করা বা সহায়তা করা', example: 'The new software will facilitate faster communication.' },
            { id: 3, category: 'formal', normal: 'Buy (কেনা)', pro: 'Procure / Purchase', meaning: 'সংগ্রহ করা বা ক্রয় করা', example: 'The department plans to procure new hardware next month.' },
            { id: 4, category: 'business', normal: 'Start (শুরু করা)', pro: 'Initiate / Commence', meaning: 'সূচনা বা আরম্ভ করা', example: 'We will initiate the project meeting tomorrow at 10 AM.' },
            { id: 5, category: 'editorial', normal: 'Show (দেখানো/প্রকাশ করা)', pro: 'Demonstrate / Exhibit', meaning: 'প্রদর্শন বা প্রমাণ করা', example: 'The quarterly report demonstrates significant improvement.' },
            { id: 6, category: 'formal', normal: 'Stop (বন্ধ করা)', pro: 'Cease / Terminate', meaning: 'স্থগিত বা সমাপ্ত করা', example: 'They decided to cease operations until further notice.' },
            { id: 7, category: 'editorial', normal: 'Change (পরিবর্তন করা)', pro: 'Alter / Modify', meaning: 'পরিমার্জন বা পরিবর্তন করা', example: 'We may need to modify the strategic plan based on feedback.' },
            { id: 8, category: 'business', normal: 'Use (ব্যবহার করা)', pro: 'Utilize / Leverage', meaning: 'সঠিক সদ্ব্যবহার করা', example: 'We should leverage our existing resources effectively.' },
            { id: 9, category: 'formal', normal: 'Tell / Give details (বলা)', pro: 'Elaborate / Disclose', meaning: 'বিস্তারিত ব্যাখ্যা করা', example: 'Could you please elaborate on your proposed timeline?' },
            { id: 10, category: 'editorial', normal: 'Make better (উন্নত করা)', pro: 'Enhance / Ameliorate', meaning: 'বৃদ্ধি বা উন্নত করা', example: 'The training session will enhance overall work productivity.' },
            { id: 11, category: 'business', normal: 'Fix (সমাধান করা)', pro: 'Rectify / Resolve', meaning: 'সংশোধন বা সমাধান করা', example: 'Our team will rectify the system error promptly.' },
            { id: 12, category: 'formal', normal: 'Give (দেওয়া)', pro: 'Provide / Allocate', meaning: 'সরবরাহ বা বরাদ্দ করা', example: 'Management allocated sufficient funds for research.' }
        ];

        let currentCategory = 'all';
        let searchQuery = '';

        function renderWords() {
            const container = document.getElementById('wordContainer');
            const noResults = document.getElementById('noResults');
            container.innerHTML = '';

            const filtered = wordsData.filter(w => {
                const matchesCategory = currentCategory === 'all' || w.category === currentCategory;
                const matchesSearch = w.normal.toLowerCase().includes(searchQuery) ||
                                      w.pro.toLowerCase().includes(searchQuery) ||
                                      w.meaning.toLowerCase().includes(searchQuery) ||
                                      w.example.toLowerCase().includes(searchQuery);
                return matchesCategory && matchesSearch;
            });

            if (filtered.length === 0) {
                noResults.classList.remove('hidden');
                return;
            } else {
                noResults.classList.add('hidden');
            }

            filtered.forEach(item => {
                const card = document.createElement('div');
                card.className = "glass-card rounded-2xl p-5 space-y-3 relative group hover:border-indigo-500/50 transition duration-300";
                card.innerHTML = `
                    <div class="flex justify-between items-start">
                        <span class="text-[10px] uppercase tracking-wider px-2 py-0.5 rounded bg-indigo-500/20 text-indigo-300 font-semibold border border-indigo-500/30">
                            ${item.category}
                        </span>
                        <button onclick="copyToClipboard('${item.pro}')" class="text-xs bg-slate-800 hover:bg-slate-700 text-slate-300 px-2.5 py-1 rounded-lg border border-slate-700 transition">
                            📋 কপি
                        </button>
                    </div>
                    <div>
                        <div class="text-xs text-rose-400 font-medium line-through">${item.normal}</div>
                        <div class="text-lg font-bold text-emerald-400">${item.pro}</div>
                        <div class="text-xs text-slate-400 mt-0.5">${item.meaning}</div>
                    </div>
                    <div class="bg-slate-900/60 p-3 rounded-xl border border-slate-800 text-xs text-slate-300 italic">
                        "${item.example}"
                    </div>
                `;
                container.appendChild(card);
            });
        }

        function filterCategory(cat) {
            currentCategory = cat;
            document.querySelectorAll('.cat-btn').forEach(btn => {
                btn.classList.remove('bg-indigo-600', 'text-white');
                btn.classList.add('bg-slate-800', 'text-slate-300');
            });
            event.target.classList.remove('bg-slate-800', 'text-slate-300');
            event.target.classList.add('bg-indigo-600', 'text-white');
            renderWords();
        }

        document.getElementById('searchInput').addEventListener('input', (e) => {
            searchQuery = e.target.value.toLowerCase().trim();
            renderWords();
        });

        function copyToClipboard(text) {
            navigator.clipboard.writeText(text);
            const toast = document.getElementById('toast');
            document.getElementById('toastMsg').innerText = `"${text}" কপি করা হয়েছে!`;
            toast.classList.remove('opacity-0', 'translate-y-10', 'pointer-events-none');
            setTimeout(() => {
                toast.classList.add('opacity-0', 'translate-y-10', 'pointer-events-none');
            }, 2000);
        }

        renderWords();
    </script>
